\end{defi}

\end{prop}

If $F/K$ is finite, $[F:K]_{s}<\infty $ (and $[F:K]_{s}$ divides $[F:K]$). 
\end{cor}

\section{Chapter 14.3 Normal Extensions}
\begin{defi}
(14.15) An algebraic extension $F/K$ is normal if for every $\alpha \in F$, the minimal polynomial $\mathrm{irr}(\alpha ,K)$ splits in $F$.
\end{defi}
\vspace{2ex}
\begin{rmk}
$F/K$ is normal if and only if the following condition holds: if $F$ contains a zero of an irreducible polynomial $p(t)\in L[t]$, then $p(t)$ splits into linear factors over $F$. 
\end{rmk}
\vspace{2ex}
\begin{rmk}
(Motivation) Let $p(t)\in K[t]$ be irreducible. Say $p(t)=\prod _{i=1}^{n}(t-\lambda _{i})\in \bar{K}[t]$. Consider a $K$-embedding $\sigma :E\hookrightarrow \bar{K}$ where $E=K(\lambda_1,\ldots ,\lambda _{n})$ is the splitting field of $p(t)$. We want to figure out the image $\sigma (E)\subset \bar{K}$. We claim that $\sigma (E)=E$. For this, observe that the image of $\lambda _{i}$ under $\sigma $ is $\alpha _{j}$ for some $j$. In other words, $\sigma $ permutes $\{\lambda_1,\lambda_2,\ldots ,\lambda_{n}\}$ ($\sigma :\{\lambda _{1},\ldots ,\lambda _{n}\}\rightarrow \{\lambda_1,\ldots ,\lambda _{n}\}$). All in all, $\sigma $ gives rise to a $K$-automorphism on $E$ when dealing with field extensions in which irreducible polynomials in it split.
\end{rmk}
\vspace{2ex}
\begin{thm}
Let $F/K$ be an algebraic extension. Fix an algebraic closure $\bar{K}$ of $K$ with $K\subset F\subset \bar{K}$. The following are equivalent:
\begin{itemize}
\item[(i)] There exists a subset $S\subset K[t]$ such that $F$ is a splitting field for $S$.
\item[(ii)] If $\sigma :F\rightarrow \bar{K}$ is a $K$-embedding, then $\sigma (F)=F$.
\item[(iii)] $F/K$ is normal.
\end{itemize}
\end{thm}
\vspace{2ex}
\begin{proof}
Assume that condition (ii) holds. To show that $F$ is normal over $K$, let $\alpha $ be an element of $F$. Suppose we are given a zero $\beta $ of $\mathrm{irr}(\alpha ,K)$ in $\bar{K}$. We wish to show that $\beta \in F$.

\bigskip

First we have a $K$-isomorphism $\sigma_0:K(\alpha )\rightarrow K(\beta )$. Using the isomorphism extension theorem, we can extend $\sigma $ to a $K$-embedding $\sigma :F\rightarrow \bar{K}$. By our assumption, $\sigma (F)=F$. In particular, $\beta =\sigma (\alpha )\in F$.

\vspace{2ex}
\begin{lem}
Let $F/K$ be an algebraic extension, and let $\sigma :F\rightarrow F$ be a $K$-embedding. Then $\sigma (F)=F$.
\end{lem}
\vspace{2ex}
\begin{proof}
Let $\alpha _{0} \in F$ be an element. Set $\mathrm{irr}(\alpha_0 ,K)=(t-\alpha_0 )(t-\alpha_1)\ldots (t-\alpha_{r})(t-\beta_1)\ldots (t-\beta_{s})\in \bar{K}[t]$ where $\alpha _{i}\in F$ and $\beta _{j}\in \bar{K}\backslash F$. Then $\sigma $ permutes $\{\alpha,\alpha_1,\ldots ,\alpha _{r}\}$ so that $\alpha_0 =\sigma(\alpha _{i}) $ for some $0\leq i\leq r$. 
\end{proof}

\begin{rmk}
Let $F/K$ be a finite extension. Then $F/K$ is normal if and only if
\[[F:K]_{s}=|\mathrm{Aut}(F/K)|\]

\end{rmk}

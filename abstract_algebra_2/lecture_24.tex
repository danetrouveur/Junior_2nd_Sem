\begin{rmk}
Last time, we considered the Galois group of a general polynomial of degree $n$. Today, we will consider $G_{f}$ for a cubic $f$.
\end{rmk}
\vspace{2ex}
\begin{recall}
We let $K$ a field, $x_1,\ldots, x_{n}$ indeterminates, and
\[g(t)=\prod^{n}_{i=1}(t-x_{i})=t^{n}-s_1t^{n-1}+s_2t^{n-2}+\ldots +(-1)^{n}s_{n}\in K(s_1,\ldots ,s_{n})\]
Define 
\[E=K(s_1,\ldots ,s_{n})\qquad\&\qquad f(t)\in E[t]\]
and $G_{g}=\mathrm{Gal}(F/E)$
\end{recall}
\vspace{2ex}
\begin{prop}
(15.37) $G_{g}$ is isomorphic to $S_{n}$.
\end{prop}
\vspace{2ex}
\begin{rmk}
Let $G_{g}\in E[t]$ be irreducible by the lemma.  
\end{rmk}
\vspace{2ex}
\begin{rmk}
Let $f(t)\in K[t]$ be a polynomial of degree $n$ and let $\alpha_1,\ldots ,\alpha _{n}$ be the zeros of $f(t)$ in $\bar{K}$.
\[\delta =\prod_{1\leq i\leq j\leq n}(a_{j}-a_{i})\qquad\&\qquad D=\delta ^2\]
is called the discriminant of $f$ in $\bar{K}$.
\end{rmk}
\vspace{2ex}
\begin{ex}
(Galois group of polynomials of degree 3) We warm up by investigating $f(t)=t^2+bt+c\in K[t]$.
\begin{itemize}
\item[(i)] If $f$ has a repeated root, $G_{f}=1$
\item[(ii)] If $f(t)=(t-\alpha )(t-\beta )$, $\alpha \ne \beta $, let $F$ be the splitting field of $f$. $F=K(\alpha ,\beta )=K(\alpha )=K(\delta )$. In this case, $\delta _{f}=\beta -\alpha $ and $D=\delta _{f}^2=(\beta -\alpha )^2=b^2-4c$.
\[G_{f}=\mathrm{Gal}(F/K)=\mathrm{Gal}(K(\delta )/K)=S_{2}\]
\end{itemize}
\end{ex}
\vspace{2ex}
\begin{rmk}
Let $g(t)=t^3-s_1t^2+s_2t-s_3$ and $D=-4s^3_{2}-27s_{3}^2-4s_{1}^3s_{3}+s_1^2s_2^2+18s_1s_2s_3$ (hint, vandermonde determinant). 
\end{rmk}
\vspace{2ex}
\begin{recall}
Let $A_{n}$ be the alternating group, the group of even permutations. Notice that $A_{n}\triangleleft S_{n}$. Also,
\[A_{n}=\mathrm{ker}(S_{n}\rightarrow \{\pm 1\})\]
\end{recall}
\vspace{2ex}
\begin{lem}
(Transposition) If $\sigma $ is a transposition, then $\sigma (\delta _{g})=-\delta _{g}$. Here, $g(t)$ is the general polynomial of degree $n$. 
\end{lem}
\vspace{2ex}
\begin{thm}
Let $n\geq 2$ be an integer. Then, 
\begin{itemize}
\item[(i)] $D_{g}\in E=K(s_1,\ldots ,s_{n})$.
\item[(ii)] If $\mathrm{char}(K)\ne 2$, $F^{A_{n}}=E(\delta _{g})$.
\end{itemize}
We have the towers $F=K(x_1,\ldots ,x_{n})-F^{A_{n}}=E(\delta _{g})-E=K(s_1,\ldots ,s_{n})$ and $1-A_{n}-\mathrm{Gal}(F/E)=S_{n}$.
\end{thm}
\vspace{2ex}
\begin{proof}
(I) Since $S_{n}$ is generated by transpositions, if $\sigma \in S_{n}$, then 
\[\sigma (D_{g})=\sigma (\delta _{g})^2=(\pm \delta _{g})^2=D_{g}\]
(II) First, observe that $\delta _{g}\in F^{A_{n}}$ by the transposition lemma. That is, $F^{A_{n}}\supset E(\delta _{g})$. Note that $\delta _{g}\notin E=F^{S_{n}}$ and that $\delta _{g}^2=D\in E$. Therefore, $[E(\delta _{g}):E]=2$. To complete the proof, we will show that $[F^{A_{n}}:E]=2$. Indeed,
\[[F^{A_{n}}:E]=\dfrac{[F:E]}{[F:F^{A_{n}}]}=\dfrac{|\mathrm{Gal}(F/E)|}{|\mathrm{Gal}(F/F^{A_{n}})|}=\dfrac{|S_{n}|}{|A_{n}|}=2\]
where we considered the diagram
\[F-F^{A_{n}}-E(\delta _{g})-E\]
\end{proof}
\vspace{2ex}
\begin{cor}
Let $K$ be a field with $\mathrm{char}(K)\ne 2$. Let $f(t)$ be an irreducible separable polynomial of degree $n\geq 2$. Let $F$ be the splitting field of $f(t)$. Then,
\begin{itemize}
\item[(i)] $F^{G_{f}\cap A_{n}}=K(\delta _{f})$.
\item[(ii)] $G_{f}\triangleleft A_{n}$ if and only if $\delta _{f}\in K$.
\end{itemize}
\end{cor}
\vspace{2ex}
\begin{ex}
(15.41) Let $K$ be a field with $\mathrm{char}(K)\ne 2$. Let $f(t)$ be an irreducible separable polynomial of degree $3$. Then,
\begin{itemize}
\item[(i)] $G_{f}\cong S_{3}$ if and only if $\delta _{f}\notin K$.
\item[(ii)] $G_{f}\cong A_3$ if and only if $\delta _{f}\in K$. 
\end{itemize}
\end{ex}

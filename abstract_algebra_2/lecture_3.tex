\begin{defi}
We let $K[\alpha ]$ denote the subring generated by $K$ and $\alpha $ and $K(\alpha )$ denote the subfield generated by $K$ and $\alpha $ as seen  below.
\[K[\alpha ]=\{f(\alpha )\in F \,|\, f(t)\in K[t]\}\]
and
\[K(\alpha )=\Big\{\dfrac{f(\alpha )}{g(\alpha )}\in F \,|\, f(t),g(t)\in K[t]\ g(\alpha )\ne 0\Big\}\]
\end{defi}
\vspace{2ex}
\begin{recall}
Take a principle ideal domain (PID) $R$ and a nonzero $p\in R$ . The following are equivalent
\begin{itemize}
	\item[(i)] $p$ is prime
	\item[(ii)] $(p)\subset R$ is prime
	\item[(iii)] $(p)\subset R$ is maximal
	\item[(iv)] $p$ is irreducible
\end{itemize}

\end{recall}

\vspace{2ex}
\begin{defi}
	$\alpha \in F$ is algebraic over $K$ if $p(\alpha )=0$ for some nonzero $p(t)\in K[t]$. Otherwise, $\alpha $ is called transcendental over $K$. 
\end{defi}
\vspace{2ex}
\begin{ex}
\begin{itemize}
	\item[(i)] $\alpha \in K\subset F$ is algebraic over $K$
	\item[(ii)] $i\in {\bm C}$ is algebraic over ${\bm R}$
	\item[(iii)] $\sqrt{3}\in {\bm R}$ is algebraic over ${\bm Q}$
\end{itemize}
\end{ex}
\vspace{2ex}
\begin{defi}
	(13.20) If $\alpha \in F$ is algebraic over $K$, the minimal polynomial of $\alpha $ over $K$ is the unique monic irreducible polynomial $p(t)\in K[t]$ such that $p(\alpha )=0$. To understand this definition, consider the evaluation homomorphism 
	\[\mathrm{ev}_{\alpha }:K[t]\rightarrow F:f(t)\mapsto f(\alpha )\]
where
\[\sum ^{n}_{i=0}\alpha_{i}t^{i}\mapsto \sum ^{n}_{i=0}a _{i}\alpha ^{i}\]
(1) Then its kernal $\mathop{\mathrm{ker}}\mathrm{ev}_{\alpha }$ is of the form $(p(t))$ for some $p(t)\in K[t]$ (because $K[t]$ is a principle ideal domain). 

\bigskip

(2) By the 1st isomorphism theorem, it follows that
\[K[t]\,/\,\mathop{\mathrm{ker}}\mathrm{ev}_{\alpha }=K[t]\,/\,(p(t))\]
is isomorphic to $\mathop{\mathrm{im}}\mathrm{ev}_{\alpha }$.

\bigskip

(3) $\mathop{\mathrm{im}}\mathrm{ev}_{\alpha }=K[\alpha ]$

\bigskip

(4) We claim that $p(t)\in K[t]$ is irreducible. Note that 
\[K[t]\,/\,(p(t))\cong \mathop{\mathrm{im}}\mathrm{ev}_{\alpha }=K[\alpha ]\subset F\]
is an integral domain (since $F$ is a field and $K[\alpha ]$ is its subring). This implies that $(p(t))$ is a prime ideal of $K[t]$. Using this fact, and the fact that $\alpha $ is algebraic, we conclude that $p(t)\in K[t]$ is irreducible.

\bigskip

(5) We may suppose that $p(t)$ is monic.

\bigskip

(6) So far, we have a monic irreducible $p(t)\in K[t]$ with $p(\alpha )=0$. The uniqueness of $p(t)$ is still required. Suppose $q(t)\in K[t]$ is a monic irreducible polynomial such that $q(\alpha )=0$. Then $q(t)\in \mathop{\mathrm{ker}}\mathrm{ev}_{\alpha }$, so that $q(t)=r(t)\cdot p(t)$ for some $r(t)\in K[t]$. Since $q(t)\in K[r]$ is irreducible and $p(t)$ is a nonunit,
\[r(t)\in (K[t])^{\times }=K^{\times }\]
Write $r(t)=r$. We deduce that $r=1$ (since both $p(t)$ and $q(t)$ are monic). 
\end{defi}
\vspace{2ex}
\begin{defi}
We let $K[\alpha ]$ denote the subring generated by $K$ and $\alpha $. We let $K(\alpha )$ denote the subfield generated by $K$ and $\alpha $.
\end{defi}
\vspace{2ex}
\begin{rmk}
Take a principle ideal domain (PID) $R$ and a nonzero $p\in R$ . The following are equivalent
\begin{itemize}
	\item[(i)] $p$ is prime
	\item[(ii)] $(p)\subset R$ is prime
	\item[(iii)] $(p)\subset R$ is maximal
	\item[(iv)] $p$ is irreducible
\end{itemize}

\end{rmk}

\[\dfrac{1}{2}\mathrm{\Sigma} ^3u^{k}_{+}(0)=\pm \dfrac{1}{2}u_{+}^{k}(0)\qquad\&\qquad \dfrac{1}{2}\mathrm{\Sigma} ^3u^{j}_{-}(0)=\pm \dfrac{1}{2}u^{j}_{-}(0)\]
which comes from the following list of facts:
\[\begin{align*}
\mathrm{(i)}\qquad\dfrac{1}{2}\begin{pmatrix}
\sigma ^{3}&0\\0&\sigma ^{3}
\end{pmatrix}u^{1}_{+}(0)=&\dfrac{1}{2}\begin{pmatrix}
\sigma ^{3}&0\\0&\sigma ^{3}
\end{pmatrix}\begin{pmatrix}
\varphi_{+}^{1}\\ \varphi ^{1}_{+}
\end{pmatrix}=\dfrac{1}{2}\begin{pmatrix}
\varphi_{+}^{1}\\ \varphi ^{1}_{+}
\end{pmatrix}\\
 \mathrm{(ii)}\qquad\dfrac{1}{2}\begin{pmatrix}
\sigma^3&0\\0&\sigma ^3
\end{pmatrix}u^{2}_{+}(0)=&\dfrac{1}{2}\begin{pmatrix}
\sigma ^3&0\\0&\sigma ^3
\end{pmatrix}\begin{pmatrix}
\varphi^{2}_{+}\\\varphi^{2}_{+}
\end{pmatrix}=-\dfrac{1}{2}\begin{pmatrix}
\varphi^{2}_{+}\\\varphi^{2}_{+}
\end{pmatrix}\\
\mathrm{(iii)}\qquad\dfrac{1}{2}\begin{pmatrix}
\sigma ^{3}&0\\0&\sigma ^{3}
\end{pmatrix}u^{1}_{-}(0)=&\dfrac{1}{2}\begin{pmatrix}
\sigma ^{3}&0\\0&\sigma ^{3}
\end{pmatrix}\begin{pmatrix}
\varphi_{-}^{1}\\ \varphi ^{1}_{-}
\end{pmatrix}=\dfrac{1}{2}\begin{pmatrix}
\varphi_{-}^{1}\\ \varphi ^{1}_{-}
\end{pmatrix}\\
 \mathrm{(iv)}\qquad\dfrac{1}{2}\begin{pmatrix}
\sigma^3&0\\0&\sigma ^3
\end{pmatrix}u^{2}_{-}(0)=&\dfrac{1}{2}\begin{pmatrix}
\sigma ^3&0\\0&\sigma ^3
\end{pmatrix}\begin{pmatrix}
\varphi^{2}_{-}\\\varphi^{2}_{-}
\end{pmatrix}=-\dfrac{1}{2}\begin{pmatrix}
\varphi^{2}_{-}\\\varphi^{2}_{-}
\end{pmatrix}
\end{align*}\]
